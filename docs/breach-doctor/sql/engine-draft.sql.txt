CREATE TABLE IF NOT EXISTS breach_diagnoses (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  issue_id INTEGER NOT NULL,                 -- the open agent_issues row; a system-level case first files a BREACH-DOCTOR-1: issue and uses its id
  case_key TEXT NOT NULL,                    -- cause-level dedupe key '<rule_id>:<subject>'; many issues may share one case_key (one action, many issues)
  family TEXT NOT NULL,                      -- enumerated by code from the title prefix (never copied text)
  rule_id TEXT NOT NULL,
  rule_version INTEGER NOT NULL DEFAULT 1,
  cause_class TEXT NOT NULL CHECK (cause_class IN ('heartbeat-insert-failing','planner-name-resolution','probe-lane-conflict','probe-author-dead','replan-no-effect','plan-gap-no-data-lane','no-measurable-dod','stale-contract','code-fixable','owner-bound','unclassified')),

  action_kind TEXT NOT NULL DEFAULT 'none' CHECK (action_kind IN ('none','restore-contract','requeue','supersede-contract','rearm-plan','ops-knob','quarantine-trigger','author-probe','code-task-issue','owner-card','close-on-probe')),

  verify_class TEXT,                         -- remediation_contracts.class of the closing probe (doctor-<id>)
  mode TEXT NOT NULL DEFAULT 'on' CHECK (mode IN ('measure','on')),

  error_code TEXT,

  diagnosed_at TEXT,
  applied_at TEXT,
  verified_at TEXT,
  last_progress_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%SZ','now')),
  lease_until TEXT,                          -- set and renewed ONLY on a progress event; never more than 120 minutes ahead (SLA-120M-1)

-- The real-owner definition (Phase B swaps it into v_issue_age_ladder.doer). owner_kind is NULL when nothing owns the issue.
-- A doctor lease counts only while it is live, backed by an artefact, and has progressed inside 120 minutes. An open IDENTITY-BOUND card counts as owner-bound
-- and is reported apart so it can never hide inside the fleet's own number.
CREATE VIEW IF NOT EXISTS v_breach_owner AS
SELECT a.id AS issue_id,
  CASE
    WHEN instr(COALESCE(a.description,''), 'code-task:') > 0 THEN 'code-task line'
    WHEN EXISTS (SELECT 1 FROM code_tasks k WHERE k.goal LIKE '[issue #' || a.id || ']%' AND k.status NOT IN ('failed','closed','superseded','needs_human')) THEN 'live code task'
    WHEN EXISTS (SELECT 1 FROM work_claims w WHERE (w.issue_id = a.id OR w.path = 'issue:' || a.id) AND w.released_at IS NULL AND w.expires_at > strftime('%Y-%m-%dT%H:%M:%SZ','now')) THEN 'live work claim'
    WHEN EXISTS (SELECT 1 FROM breach_diagnoses d WHERE d.issue_id = a.id AND d.mode = 'on' AND d.state IN ('queued','applied','verifying')
                   AND d.action_ref IS NOT NULL AND d.action_kind <> 'none'
                   AND d.lease_until > strftime('%Y-%m-%dT%H:%M:%SZ','now')
                   AND d.last_progress_at > strftime('%Y-%m-%dT%H:%M:%SZ','now','-120 minutes')) THEN 'breach doctor'
    WHEN EXISTS (SELECT 1 FROM breach_diagnoses d WHERE d.issue_id = a.id AND d.state = 'handed-owner' AND d.action_kind = 'owner-card'
                   AND EXISTS (SELECT 1 FROM human_actions h WHERE h.slug = d.action_ref AND h.status = 'open' AND instr(COALESCE(h.why,''), 'IDENTITY-BOUND:') > 0)) THEN 'owner card (identity-bound)'
    ELSE NULL END AS owner_kind
FROM agent_issues a WHERE a.status = 'open';

CREATE VIEW IF NOT EXISTS v_breach_doctor_status AS
SELECT
  COALESCE((SELECT value FROM ops_config WHERE key = 'breach_doctor'), 'measure') AS mode,
  COALESCE((SELECT value FROM ops_config WHERE key = 'breach_doctor_ai'), 'on') AS ai,
  (SELECT MAX(ts) FROM breach_diagnosis_events WHERE event = 'tick') AS last_tick_at,
  (SELECT COUNT(*) FROM breach_diagnoses WHERE opened_at > strftime('%Y-%m-%dT%H:%M:%SZ','now','-24 hours')) AS cases_24h,
  (SELECT COUNT(*) FROM breach_diagnoses WHERE state = 'verified' AND verified_at > strftime('%Y-%m-%dT%H:%M:%SZ','now','-24 hours')) AS verified_24h,
  (SELECT COUNT(*) FROM breach_diagnoses WHERE state IN ('failed','reverted') AND last_progress_at > strftime('%Y-%m-%dT%H:%M:%SZ','now','-24 hours')) AS failed_24h,
  (SELECT COUNT(*) FROM breach_diagnoses WHERE state IN ('queued','applied','verifying')) AS in_flight,
  (SELECT COUNT(*) FROM breach_diagnoses WHERE state IN ('queued','applied','verifying') AND lease_until < strftime('%Y-%m-%dT%H:%M:%SZ','now')) AS lease_expired,
  (SELECT COUNT(*) FROM breach_diagnosis_events WHERE event = 'taint-drop' AND ts > strftime('%Y-%m-%dT%H:%M:%SZ','now','-24 hours')) AS taint_drops_24h;
