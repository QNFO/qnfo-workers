-- CM-R10: the session-only queue. A read model over rows the doctor handed over; no new table.
-- Marker conventions already in force: a "session-task:" line in agent_issues.description and issue_triage.owner LIKE 'session%'
-- (both counted by v_autonomy_first_audit.session_dependent_issues_open). The doctor adds the exact blocker in issue_triage.remediation.
CREATE VIEW IF NOT EXISTS v_session_queue AS
SELECT a.id AS issue_id,
       a.priority,
       ROUND((CAST(strftime('%s','now') AS INTEGER) * 1000 - a.created_at) / 3600000.0, 1) AS age_h,
       t.remediation AS blocker,
       substr(a.description, instr(a.description, 'session-task:'), 160) AS session_task_line,
       (SELECT w.holder FROM work_claims w WHERE w.path = 'issue:' || a.id AND w.released_at IS NULL AND w.expires_at > strftime('%Y-%m-%dT%H:%M:%SZ','now') LIMIT 1) AS claimed_by
FROM agent_issues a
JOIN issue_triage t ON t.issue_id = a.id
WHERE a.status = 'open' AND t.owner = 'session' AND t.remediation LIKE 'BLOCKER %'
ORDER BY CASE a.priority WHEN 'critical' THEN 0 WHEN 'high' THEN 1 WHEN 'medium' THEN 2 WHEN 'low' THEN 3 ELSE 4 END, a.created_at;
