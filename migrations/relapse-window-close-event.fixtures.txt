#@ view v_closure_relapses v_closure_relapses 1708
CREATE VIEW v_closure_relapses AS
SELECT x.issue_id, x.class, x.status, x.closed_ms, x.close_evidence, x.fails, x.last_observed, x.max_attempts,
  (SELECT MIN(o.id) FROM agent_issues o WHERE o.status = 'open' AND o.id <> x.issue_id AND lower(trim(o.title)) = lower(trim(x.title))) AS twin_issue_id
FROM (
  SELECT c.issue_id, c.class, a.status, a.title, c.max_attempts, t.close_evidence, a.updated_at AS closed_ms,
    (SELECT COUNT(*) FROM remediation_verifications v WHERE v.class = c.class AND v.pass = 0
       AND COALESCE(v.observed, '') <> '' AND lower(trim(v.observed)) NOT LIKE 'pending%'
       AND v.verified_at > MAX(datetime(CASE WHEN a.updated_at > 100000000000 THEN a.updated_at / 1000 ELSE a.updated_at END, 'unixepoch'),
             COALESCE((SELECT MAX(p.verified_at) FROM remediation_verifications p WHERE p.class = c.class AND p.pass = 1), '1970-01-01'))) AS fails,
    (SELECT substr(v.observed, 1, 120) FROM remediation_verifications v WHERE v.class = c.class ORDER BY v.verified_at DESC LIMIT 1) AS last_observed
  FROM remediation_contracts c JOIN agent_issues a ON a.id = c.issue_id LEFT JOIN issue_triage t ON t.issue_id = a.id
  WHERE c.status IN ('active', 'holding') AND a.status IN ('closed', 'resolved') AND typeof(a.updated_at) = 'integer'
    AND CASE WHEN a.updated_at > 100000000000 THEN a.updated_at / 1000 ELSE a.updated_at END < CAST(strftime('%s', 'now') AS INTEGER) - 7200
    AND lower(substr(COALESCE(t.close_evidence, ''), 1, 10)) <> 'superseded' AND lower(substr(COALESCE(t.close_evidence, ''), 1, 4)) <> 'moot'
) x
WHERE x.fails >= MAX(1, COALESCE(x.max_attempts, 3))
  AND COALESCE(x.last_observed, '') <> '' AND lower(trim(x.last_observed)) NOT LIKE 'pending%'

#@ trigger decision_log decision_predictions_parse 2040
CREATE TRIGGER decision_predictions_parse AFTER INSERT ON decision_log
WHEN instr(COALESCE(NEW.predicted_outcome, ''), ' within ') > 0
BEGIN
  INSERT OR IGNORE INTO decision_predictions (decision_id, kind, subject, op, value, due_at, origin)
    SELECT NEW.id, 'metric', t.tok1, t.tok2, CAST(t.tok3 AS REAL),
      strftime('%Y-%m-%dT%H:%M:%SZ', 'now', '+' || CAST(t.after AS INTEGER) || CASE WHEN instr(t.after, ' d') > 0 THEN ' days' ELSE ' hours' END), 'parsed'
    FROM (
      SELECT u.tok1, u.tok2,
        CASE WHEN instr(u.rest2, ' ') > 0 THEN substr(u.rest2, 1, instr(u.rest2, ' ') - 1) ELSE u.rest2 END AS tok3, u.after
      FROM (
        SELECT v.tok1,
          CASE WHEN instr(v.rest1, ' ') > 0 THEN substr(v.rest1, 1, instr(v.rest1, ' ') - 1) ELSE v.rest1 END AS tok2,
          CASE WHEN instr(v.rest1, ' ') > 0 THEN ltrim(substr(v.rest1, instr(v.rest1, ' ') + 1)) ELSE '' END AS rest2, v.after
        FROM (
          SELECT substr(w.p, 1, instr(w.p, ' ') - 1) AS tok1, ltrim(substr(w.p, instr(w.p, ' ') + 1)) AS rest1,
            ltrim(substr(w.p, instr(w.p, ' within ') + 8)) AS after
          FROM (SELECT trim(NEW.predicted_outcome) AS p) w
        ) v
      ) u
    ) t
    WHERE t.tok1 IN (SELECT metric FROM metric_registry) AND t.tok2 IN ('<', '<=', '>', '>=', '=')
      AND t.tok3 GLOB '[0-9]*' AND CAST(t.after AS INTEGER) > 0;
  INSERT OR IGNORE INTO decision_predictions (decision_id, kind, subject, op, value, due_at, origin)
    SELECT NEW.id, 'issue', CAST(CAST(substr(w.p, 7) AS INTEGER) AS TEXT), 'closed', NULL,
      strftime('%Y-%m-%dT%H:%M:%SZ', 'now', '+' || CAST(w.after AS INTEGER) || CASE WHEN instr(w.after, ' d') > 0 THEN ' days' ELSE ' hours' END), 'parsed'
    FROM (SELECT trim(NEW.predicted_outcome) AS p, ltrim(substr(trim(NEW.predicted_outcome), instr(trim(NEW.predicted_outcome), ' within ') + 8)) AS after) w
    WHERE lower(substr(w.p, 1, 6)) = 'issue ' AND CAST(substr(w.p, 7) AS INTEGER) > 0
      AND instr(lower(w.p), ' closed within ') > 0 AND CAST(w.after AS INTEGER) > 0;
END
